== 配置文件

./xxl-job-admin/application.yml
[source,yaml,%linenums]
----
xxl:
  job:
    # 调度中心通讯TOKEN [选填]：非空时启用
    accessToken:
spring:
  # 数据库连接
  datasource:
    url: jdbc:mysql://mysql:3306/xxljob?useUnicode=true&characterEncoding=UTF-8&autoReconnect=true&serverTimezone=Asia/Shanghai
    # 连接数据库的用户，可根据实际情况设置
    username: xxljob
    # 连接数据库的密码，可根据实际情况设置
    password: xxljob
  # 报警发件人邮箱
  mail:
    # 发件邮箱服务器，具体可以通过网页登录邮箱查看
    host: smtp.qq.com
    # 发件邮箱服务器端口号，具体可以通过网页登录邮箱查看
    port: 25
    # 发件人邮箱
    username: xxx@qq.com
    # 这里的密码是授权码，可以通过网页登录邮箱设置
    password: xxxxxxxx

  ....
----

== 单机

[source,sh]
----
mkdir -p ~/opt/xxljob/
----

.~/opt/xxljob/docker-compose.yml
[source,yaml,%linenums]
----
services:
  xxljob:
    image: xuxueli/xxl-job-admin:3.4.2
    hostname: xxljob
    container_name: xxljob
    ports:
      - "7788:8080"
    environment:
      PARAMS: >-
        --spring.datasource.url=jdbc:mysql://mysql:3306/xxljob?useUnicode=true&characterEncoding=UTF-8&autoReconnect=true&serverTimezone=Asia/Shanghai
        --spring.datasource.username=xxljob
        --spring.datasource.password=xxljob
    volumes:
      - data:/data/applogs
    restart: unless-stopped

volumes:
  data:

networks:
  default:
    external: true
    name: rebue
----

* 部署

[source,shell]
----
docker compose -f ~/opt/xxljob/docker-compose.yml up -d
----
